# School Management System (SMS) — Fresh Start

A new starter project built with Next.js 14, TypeScript, and Supabase Auth.

## Current stage
- Responsive overview and login UI.
- Supabase browser and server clients using the public anon/publishable key.
- Email/password login flow.
- Protected `/dashboard` route and server-side profile lookup.
- Server sign-out route.
- Super Admin Schools page with school directory and a school-registration form.
- Principal student-registration/search, class-creation/list, attendance-recording/list, result entry/Principal approval, fee creation/payment updates, Principal teacher invitation, and parent invitations with multi-child linking connected to Supabase; database RLS remains the final access boundary.
- Initial school messaging and PTA/event reminder forms that save message/event and recipient records; automatic delivery is not enabled yet.
- Principal invitation form linked to an active school.
- `supabase/functions/create-principal/index.ts` contains the protected Edge Function used to create and link Principal accounts; the function has been deployed to the connected Supabase project.

## Setup
1. Use Node.js 20 or later.
2. Copy `.env.example` to `.env.local`.
3. Add your Supabase project URL and anon/publishable key.
4. Do not run `supabase/migrations/001_initial_schema.sql` against the existing SCHOOL MANAGEMENT SYSTEM Supabase project. The live database schema was created by separate migrations and is not identical to this starter SQL.
5. Run `npm install`, then `npm run dev`.

## Important security notes
- This starter does not include or need a service-role key.
- Never put a service-role key in `NEXT_PUBLIC_*` variables or browser code.
- The SQL migration is a starting draft and must be reviewed/tested against a new, empty Supabase project before use. Do not run it in a database containing existing data.
- Principal invitation flow is coded and the Edge Function is deployed, but end-to-end invitation email delivery and profile creation have not yet been tested with a real invitation. Auth email configuration must be checked before production use. Full per-role permissions, parent-child linking, and runtime security testing are not complete yet.
- A production Next.js build has not been verified in this environment because dependencies could not be installed within the available network/time limit.
- Do not enter real credentials until Supabase is configured and the app has been deployed/tested in a trusted environment.

import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request });
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;

  // Preview mode allows the public overview/login UI to run without credentials.
  // Protected routes fail closed if Supabase is not configured.
  if (!url || !key) {
    if (request.nextUrl.pathname.startsWith("/dashboard")) {
      const target = new URL("/login", request.url);
      target.searchParams.set("next", request.nextUrl.pathname);
      return NextResponse.redirect(target);
    }
    return response;
  }

  const supabase = createServerClient(url, key, {
    cookies: {
      getAll() { return request.cookies.getAll(); },
      setAll(cookiesToSet) {
        cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
        response = NextResponse.next({ request });
        cookiesToSet.forEach(({ name, value, options }) => response.cookies.set(name, value, options));
      },
    },
  });

  const { data: { user } } = await supabase.auth.getUser();
  const path = request.nextUrl.pathname;
  if (path.startsWith("/dashboard") && !user) {
    const target = new URL("/login", request.url);
    target.searchParams.set("next", path);
    return NextResponse.redirect(target);
  }
  if ((path === "/login") && user) return NextResponse.redirect(new URL("/dashboard", request.url));
  return response;
}

export const config = { matcher: ["/dashboard/:path*", "/login"] };

# SMS Feature and Role Responsibility Matrix

This is the acceptance checklist for the School Management System. A feature is **not complete** merely because its menu or page exists. It is complete only after the UI, server-side authorization, database/RLS policies, and tests all work together.

## 1. Role responsibilities

| Feature / duty | Super Admin | Principal | Teacher | Parent | Acceptance rule |
|---|---|---|---|---|---|
| Create and manage schools | Full platform access | Own school only: view school details | No | No | A user cannot read or change another school's records. |
| Create/invite Principal accounts | Yes; assign to an active school | No | No | No | Principal profile is linked to exactly the authorized school; invitation and first login are tested. |
| Manage teachers | Platform support only when explicitly authorized | Add/edit/deactivate teachers in own school; issue unique teacher code | View own profile; cannot assign roles or school | No | Teacher cannot self-promote or change school/role. |
| Teacher first-time access | No routine access to teacher credentials | Issue/reissue teacher code | Redeem code, verify identity and set own password | No | Codes are unique, expire or become unusable after redemption, and are never treated as a role-selection substitute. |
| Manage students | Platform support only when explicitly authorized | Register/search/update students in own school; system creates unique student number | View only students in assigned classes | View only linked children | Student numbers are unique; access is scoped by school/class/parent-child link. |
| Classes and subjects | Platform oversight only | Create classes, assign teachers and subjects | View assigned classes/subjects | View only information necessary for linked children | Teachers cannot access unassigned classes. |
| Attendance | No routine school-record editing | View/manage school attendance | Mark and view attendance only for assigned classes | View attendance for linked children | Server/database checks enforce class and child scope. |
| Exams and results | No routine result editing | Manage exam setup and school results; authorized corrections | Enter/update results only for assigned classes and permitted periods | View results for linked children; cannot edit | Result changes are authorized and audited; parents/teachers cannot alter protected school settings. |
| Fees and payments | Platform oversight only; no automatic access to school finance records | Set/record/verify fees and payments for own school | No fee/result override privileges by default | View own children's fee balances/receipts and payment status | Only authorized Principal/finance permissions can change financial records; parent sees only linked children. |
| Messages | Platform announcements only if explicitly implemented | Message all school users or selected teachers/recipients | Send/receive permitted school messages | Receive school/child-related messages; reply only if enabled | Recipient selection and school boundaries are enforced on the server. |
| PTA/events/reminders | Platform-level only if explicitly implemented | Create reminder/event, choose date/time and parent recipients | View relevant notices | Receive reminders for their linked children | A reminder such as “PTA 12/04/2027” is stored with date, recipients, delivery status, and timezone rules; sending is tested. |
| Parent-child linking | No routine linking without authorized workflow | Register/link a parent to one or more students | No | View each linked child using the authorized parent account | One parent can access multiple linked children; a parent cannot guess a code to access an unrelated child. |
| Reports/search | Platform reports only within authorized platform scope | Search students and view school-wide reports | Reports limited to assigned classes | Reports limited to linked children | Search and exports obey the same permissions as on-screen records. |
| Account/profile | Manage platform roles and school assignment | Manage own profile and authorized school accounts | Manage own permitted profile/password | Manage own permitted profile/password | No non-Super-Admin can change their role or school assignment. |
| Audit/security | Review platform security events where available | Review school-level audit records where available | No access to restricted audit records | No access to restricted audit records | Sensitive edits record actor, action, record, and timestamp. |

## 2. Required screens and workflows

- Login, sign-out, account recovery, active-account checks, and role-based dashboard/navigation.
- Super Admin: school directory/create school, Principal invitation, platform overview, and any subscription/report/settings screen only if those are confirmed project requirements.
- Principal: school overview, student register/search, teacher register/invitation, class/subject setup, attendance, exams/results, fees/payments, messages, PTA/events/reminders, and school reports.
- Teacher: assigned classes, class roster, attendance entry, permitted results entry, and permitted messages.
- Parent: linked-children selector, each child's attendance/results/fees, and school messages/reminders.
- Responsive layouts for phone and computer.

## 3. Security and acceptance tests required before calling the project complete

1. Sign in as each of the four roles using separate test accounts and verify the correct dashboard and navigation.
2. Attempt direct URL access to every module with each role; forbidden pages/actions must be denied server-side, not merely hidden from the menu.
3. Verify two schools cannot read or change each other's students, teachers, classes, attendance, results, fees, messages, or reminders.
4. Verify a teacher cannot access an unassigned class or change their role/school, fee records, or protected school settings.
5. Verify a parent cannot access another family's child by changing a URL, ID, or student code, and can switch among all of their own linked children.
6. Verify unique student and teacher codes, first-use redemption, invalid/used code rejection, and password setup.
7. Verify Principal can search students, message all or selected recipients, and schedule PTA reminders for selected parents; confirm actual delivery status rather than only a saved form.
8. Verify results and fees are editable only by authorized roles; record/audit sensitive changes.
9. Run database/RLS tests, lint/type-check, a production build, and phone/desktop browser tests. Resolve all critical/high security findings or document and approve remaining risks.
10. Test account recovery and invitation emails with the configured Supabase email provider before production.

## 4. Current implementation status (verified as of 2026-10-20, latest working session)

- **Present in code:** login flow, protected dashboard shell, role-specific navigation, Super Admin school page/form, Principal invitation form, deployed `create-principal` Edge Function, role-specific dashboard count queries, student registration/search UI, class creation/list UI, attendance recording/list UI, staff message form, PTA/event reminder form, and Principal teacher-invitation form with a unique staff number, and parent invitation/child-link form.
- **Partial / not end-to-end tested:** school creation and Principal invitation; invitation email delivery and profile creation have not been exercised with a real test account. Dashboard live counts, student/class/attendance forms, message recipient selection, and reminder forms have source code but have not been verified in a running browser with separate role accounts.
- **Not finished:** teacher first-time access-code redemption (teacher invitation/staff number is only an identifier; code-based password setup is not complete), automated browser tests of subject and parent workflows with separate school-role accounts, exam setup beyond subject/term/year result entry, full fee/payment workflow, parent/teacher invitation email delivery tests, delivery of messages/reminders (email/SMS provider and scheduler), real audit event capture/UI, reports, and the full role-by-role automated/manual test suite. Student/class/attendance/result-entry and Principal result-approval UI now have initial database-backed workflows. Parent result visibility is restricted to approved rows at the database level; edit/deactivate handling and end-to-end tests remain.
- **Database update:** additive migrations are applied to live Supabase for school-scoped unique student/staff codes, messaging, event/reminder recipient records, audit log storage, subject catalogue and class-subject assignments, and RLS policies. Live table listing confirms `subjects` and `class_subjects` exist with RLS enabled. A rolled-back SQL test confirmed cross-school and duplicate subject assignments are blocked. No real school, parent, teacher, or student test data was created.

- **Build/runtime:** TypeScript/TSX syntax transpilation passed for 19 files, but dependency installation has not succeeded in this environment; a full TypeScript check, production build, and browser/device test have not passed yet.

Do not mark any item complete until its acceptance rule and relevant tests pass against the live app and database.

"use client";

import { FormEvent, useState } from "react";
import { useRouter } from "next/navigation";
import { createSupabaseBrowserClient } from "@/lib/supabase/client";

type SchoolOption = { id: string; name: string; is_active: boolean };

export default function PrincipalCreateForm({ schools }: { schools: SchoolOption[] }) {
  const router = useRouter();
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const formElement = event.currentTarget;
    setBusy(true);
    setError("");
    setSuccess("");
    const form = new FormData(formElement);
    const fullName = String(form.get("full_name") ?? "").trim();
    const email = String(form.get("email") ?? "").trim().toLowerCase();
    const schoolId = String(form.get("school_id") ?? "").trim();
    if (fullName.length < 2 || !email || !schoolId) {
      setError("Enter the Principal's name, email, and select a school.");
      setBusy(false);
      return;
    }
    try {
      const supabase = createSupabaseBrowserClient();
      const { data, error: invokeError } = await supabase.functions.invoke("create-principal", {
        body: { full_name: fullName, email, school_id: schoolId },
      });
      if (invokeError) throw new Error("The secure account service could not be reached. Confirm that the function is deployed, then try again.");
      if (!data?.success) throw new Error(typeof data?.error === "string" ? data.error : "The Principal account could not be created.");
      formElement.reset();
      setSuccess(`Invitation sent to ${email}. The account is linked to ${data.principal.school_name}.`);
      router.refresh();
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : "Unable to create the Principal account. Please try again.");
    } finally {
      setBusy(false);
    }
  }

  const activeSchools = schools.filter((school) => school.is_active);
  return (
    <form onSubmit={submit}>
      <div className="field"><label htmlFor="principal-name">Principal full name *</label><input id="principal-name" name="full_name" required minLength={2} maxLength={160} /></div>
      <div className="field"><label htmlFor="principal-email">Principal email *</label><input id="principal-email" name="email" type="email" required maxLength={254} /></div>
      <div className="field"><label htmlFor="principal-school">School *</label><select id="principal-school" name="school_id" required defaultValue=""><option value="" disabled>Select a school</option>{activeSchools.map((school) => <option key={school.id} value={school.id}>{school.name}</option>)}</select></div>
      {activeSchools.length === 0 && <p className="sub">Register and activate a school before creating its Principal account.</p>}
      {error && <p role="alert" style={{ color: "#b42318", fontSize: 12 }}>{error}</p>}
      {success && <p role="status" style={{ color: "#067647", fontSize: 12 }}>{success}</p>}
      <button className="btn" type="submit" disabled={busy || activeSchools.length === 0}>{busy ? "Creating account…" : "Invite Principal"}</button>
    </form>
  );
}

"use client";

import { FormEvent, useState } from "react";
import { useRouter, useSearchParams } from "next/navigation";
import { createSupabaseBrowserClient } from "@/lib/supabase/client";

export default function LoginForm() {
  const router = useRouter();
  const params = useSearchParams();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    setBusy(true); setError("");
    try {
      const supabase = createSupabaseBrowserClient();
      const { error: authError } = await supabase.auth.signInWithPassword({ email, password });
      if (authError) throw authError;
      const next = params.get("next");
      router.replace(next && next.startsWith("/") && !next.startsWith("//") ? next : "/dashboard");
      router.refresh();
    } catch (e) {
      setError(e instanceof Error ? e.message : "Unable to sign in. Please try again.");
    } finally { setBusy(false); }
  }

  return <form onSubmit={submit}>
    <div className="field"><label htmlFor="email">Email address</label><input id="email" type="email" value={email} onChange={e=>setEmail(e.target.value)} autoComplete="email" required /></div>
    <div className="field"><label htmlFor="password">Password</label><input id="password" type="password" value={password} onChange={e=>setPassword(e.target.value)} autoComplete="current-password" required /></div>
    {error && <p role="alert" style={{color:"#b42318",fontSize:12}}>{error}</p>}
    <button className="btn" style={{width:"100%",marginTop:10}} type="submit" disabled={busy}>{busy ? "Signing in…" : "Sign in securely"}</button>
  </form>;
}

import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createSupabaseServerClient() {
  const cookieStore = cookies();
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const key = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
  if (!url || !key) throw new Error("Supabase environment variables are missing.");

  return createServerClient(url, key, {
    cookies: {
      get(name: string) { return cookieStore.get(name)?.value; },
      set(name: string, value: string, options: Record<string, unknown>) {
        try { cookieStore.set({ name, value, ...(options as object) }); } catch {}
      },
      remove(name: string, options: Record<string, unknown>) {
        try { cookieStore.set({ name, value: "", ...(options as object), maxAge: 0 }); } catch {}
      },
    },
  });
}

import { createClient } from "npm:@supabase/supabase-js@2.49.1";
const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS" };
function json(status: number, body: Record<string, unknown>) { return new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" } }); }
Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: corsHeaders });
  if (req.method !== "POST") return json(405, { error: "Method not allowed." });
  const authorization = req.headers.get("Authorization");
  if (!authorization?.startsWith("Bearer ")) return json(401, { error: "Sign in as Principal first." });
  const url = Deno.env.get("SUPABASE_URL"), anon = Deno.env.get("SUPABASE_ANON_KEY"), service = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
  if (!url || !anon || !service) return json(500, { error: "Server authentication is not configured." });
  const auth = createClient(url, anon, { auth: { persistSession: false, autoRefreshToken: false } });
  const { data: authData, error: authError } = await auth.auth.getUser(authorization.slice(7));
  if (authError || !authData.user) return json(401, { error: "Your session is invalid or expired." });
  const admin = createClient(url, service, { auth: { persistSession: false, autoRefreshToken: false } });
  const { data: caller, error: callerError } = await admin.from("profiles").select("id,role,school_id,is_active").eq("id", authData.user.id).maybeSingle();
  if (callerError || !caller || caller.role !== "principal" || !caller.is_active || !caller.school_id) return json(403, { error: "Only an active Principal linked to a school can add parents." });
  let body: Record<string, unknown>; try { body = await req.json(); } catch { return json(400, { error: "Invalid request body." }); }
  const email = typeof body.email === "string" ? body.email.trim().toLowerCase() : "";
  const fullName = typeof body.full_name === "string" ? body.full_name.trim() : "";
  const studentIds = Array.isArray(body.student_ids) ? [...new Set(body.student_ids.filter((x): x is string => typeof x === "string"))] : [];
  if (email.length < 5 || email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return json(400, { error: "Enter a valid parent email address." });
  if (fullName.length < 2 || fullName.length > 160) return json(400, { error: "Enter the parent's full name." });
  if (!studentIds.length || studentIds.length > 30) return json(400, { error: "Select at least one child (up to 30 children per parent account)." });
  const { data: students, error: studentError } = await admin.from("students").select("id").eq("school_id", caller.school_id).in("id", studentIds);
  if (studentError || !students || students.length !== studentIds.length) return json(403, { error: "One or more selected children are not in your school." });
  const { data: existing } = await admin.from("profiles").select("id").eq("email", email).maybeSingle();
  if (existing) return json(409, { error: "A profile already uses this email address." });
  const { data: invited, error: inviteError } = await admin.auth.admin.inviteUserByEmail(email, { data: { full_name: fullName, invited_role: "parent", school_id: caller.school_id } });
  if (inviteError || !invited.user) { const duplicate = /already|registered|exists/i.test(inviteError?.message ?? ""); return json(duplicate ? 409 : 400, { error: duplicate ? "This email may already have an Auth account. Check Authentication → Users." : "Invitation could not be sent. Check Auth email configuration." }); }
  const { error: profileError } = await admin.from("profiles").insert({ id: invited.user.id, school_id: caller.school_id, full_name: fullName, email, role: "parent", is_active: true });
  if (profileError) { await admin.auth.admin.deleteUser(invited.user.id); return json(500, { error: "Could not create parent profile; incomplete account was rolled back." }); }
  const { data: parent, error: parentError } = await admin.from("parents").insert({ school_id: caller.school_id, profile_id: invited.user.id }).select("id").single();
  if (parentError || !parent) { await admin.from("profiles").delete().eq("id", invited.user.id); await admin.auth.admin.deleteUser(invited.user.id); return json(500, { error: "Could not create parent record; incomplete account was rolled back." }); }
  const links = studentIds.map(student_id => ({ school_id: caller.school_id, parent_id: parent.id, student_id, relationship: "guardian" }));
  const { error: linkError } = await admin.from("parent_students").insert(links);
  if (linkError) { await admin.from("parents").delete().eq("id", parent.id); await admin.from("profiles").delete().eq("id", invited.user.id); await admin.auth.admin.deleteUser(invited.user.id); return json(500, { error: "Could not link the selected children; incomplete account was rolled back." }); }
  return json(201, { success: true, message: "Parent invitation sent and selected children linked.", parent: { id: invited.user.id, full_name: fullName, email, child_count: studentIds.length } });
});

    auth: { persistSession: false, autoRefreshToken: false },
  });
  const { data: authData, error: authError } = await authClient.auth.getUser(token);
  if (authError || !authData.user) return json(401, { error: "Your session is invalid or expired. Sign in again." });

  const admin = createClient(supabaseUrl, serviceRoleKey, {
    auth: { persistSession: false, autoRefreshToken: false },
  });
  const { data: caller, error: callerError } = await admin
    .from("profiles")
    .select("role, is_active")
    .eq("id", authData.user.id)
    .maybeSingle();
  if (callerError || !caller || caller.role !== "super_admin" || !caller.is_active) {
    return json(403, { error: "Only an active Super Admin can create a Principal account." });
  }

  let body: Record<string, unknown>;
  try {
    body = await req.json();
  } catch {
    return json(400, { error: "Invalid request body." });
  }

  const email = typeof body.email === "string" ? body.email.trim().toLowerCase() : "";
  const fullName = typeof body.full_name === "string" ? body.full_name.trim() : "";
  const schoolId = typeof body.school_id === "string" ? body.school_id.trim() : "";
  if (email.length < 5 || email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
    return json(400, { error: "Enter a valid Principal email address." });
  }
  if (fullName.length < 2 || fullName.length > 160) return json(400, { error: "Enter the Principal's full name." });
  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(schoolId)) {
    return json(400, { error: "Choose a valid school." });
  }

  const { data: school, error: schoolError } = await admin
    .from("schools").select("id, name, is_active").eq("id", schoolId).maybeSingle();
  if (schoolError || !school) return json(404, { error: "The selected school was not found." });
  if (!school.is_active) return json(409, { error: "This school is inactive. Activate it before adding a Principal." });

  const { data: existingProfile } = await admin.from("profiles").select("id").eq("email", email).maybeSingle();
  if (existingProfile) return json(409, { error: "A profile already uses this email address." });

  const { data: invited, error: inviteError } = await admin.auth.admin.inviteUserByEmail(email, {
    data: { full_name: fullName, invited_role: "principal", school_id: schoolId },
    redirectTo: undefined,
  });
  if (inviteError || !invited.user) {
    const message = inviteError?.message ?? "Could not send the invitation.";
    const status = /already|registered|exists/i.test(message) ? 409 : 400;
    return json(status, { error: status === 409 ? "This email may already have an account. Check Authentication → Users." : "The invitation could not be sent. Check the Auth email configuration and try again." });
  }

  const { error: profileError } = await admin.from("profiles").insert({
    id: invited.user.id,
    school_id: schoolId,
    full_name: fullName,
    email,
    role: "principal",
    is_active: true,
  });
  if (profileError) {
    await admin.auth.admin.deleteUser(invited.user.id);
    return json(500, { error: "The account could not be linked to the school. No active Principal account was kept. Please try again." });
  }

  return json(201, {
    success: true,
    message: "Principal invitation sent and account linked to the selected school.",
    principal: { id: invited.user.id, full_name: fullName, email, school_id: schoolId, school_name: school.name },
  });

  const authorization = req.headers.get("Authorization");

  const url = Deno.env.get("SUPABASE_URL");
  const anon = Deno.env.get("SUPABASE_ANON_KEY");
  const service = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  const { data: authData, error: authError } = await auth.auth.getUser(authorization.slice(7));
  if (authError || !authData.user) return json(401, { error: "Your session is invalid or expired." });

  const { data: caller, error: callerError } = await admin.from("profiles").select("id,role,school_id,is_active").eq("id", authData.user.id).maybeSingle();
  if (callerError || !caller || caller.role !== "principal" || !caller.is_active || !caller.school_id) return json(403, { error: "Only an active Principal linked to a school can invite teachers." });
  let body: Record<string, unknown>;
  try { body = await req.json(); } catch { return json(400, { error: "Invalid request body." }); }
  const email = typeof body.email === "string" ? body.email.trim().toLowerCase() : "";
  const fullName = typeof body.full_name === "string" ? body.full_name.trim() : "";
  if (email.length < 5 || email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return json(400, { error: "Enter a valid teacher email address." });
  if (fullName.length < 2 || fullName.length > 160) return json(400, { error: "Enter the teacher's full name." });
  const { data: existing } = await admin.from("profiles").select("id").eq("email", email).maybeSingle();
  if (existing) return json(409, { error: "A profile already uses this email address." });
  const staffNumber = "TCH-" + crypto.randomUUID().replaceAll("-", "").slice(0, 10).toUpperCase();
  const { data: invited, error: inviteError } = await admin.auth.admin.inviteUserByEmail(email, { data: { full_name: fullName, invited_role: "teacher", school_id: caller.school_id } });
  if (inviteError || !invited.user) {
    const duplicate = /already|registered|exists/i.test(inviteError?.message ?? "");
    return json(duplicate ? 409 : 400, { error: duplicate ? "This email may already have an Auth account. Check Authentication → Users." : "Invitation could not be sent. Check Auth email configuration." });
  }
  const { error: profileError } = await admin.from("profiles").insert({ id: invited.user.id, school_id: caller.school_id, full_name: fullName, email, role: "teacher", is_active: true });
  if (profileError) {
    await admin.auth.admin.deleteUser(invited.user.id);
    return json(500, { error: "Could not create the teacher profile. No active teacher account was kept." });
  }
  const { error: teacherError } = await admin.from("teachers").insert({ school_id: caller.school_id, profile_id: invited.user.id, staff_number: staffNumber, employment_status: "active" });
  if (teacherError) {
    await admin.from("profiles").delete().eq("id", invited.user.id);
    await admin.auth.admin.deleteUser(invited.user.id);
    return json(500, { error: "Could not create the teacher record. The incomplete account was rolled back." });
  }
  return json(201, { success: true, message: "Teacher invitation sent and teacher record created.", teacher: { id: invited.user.id, full_name: fullName, email, staff_number: staffNumber, school_id: caller.school_id } });
});

RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path TO '' AS $function$
BEGIN
  IF auth.uid() IS NOT NULL
     AND public.current_user_role() IS DISTINCT FROM 'super_admin' THEN

       OR NEW.school_id IS DISTINCT FROM OLD.school_id THEN

END;

create or replace function public.validate_class_subject_assignment()
returns trigger language plpgsql security invoker set search_path = '' as $$
begin
  if not exists (select 1 from public.classes c where c.id = new.class_id and c.school_id = new.school_id) then
    raise exception 'Class must belong to the same school.';
  end if;
  if not exists (select 1 from public.subjects s where s.id = new.subject_id and s.school_id = new.school_id and s.is_active = true) then
    raise exception 'Subject must be active and belong to the same school.';
  end if;
  if new.teacher_profile_id is not null and not exists (
    select 1 from public.profiles p join public.teachers t on t.profile_id = p.id and t.school_id = p.school_id
    where p.id = new.teacher_profile_id and p.school_id = new.school_id and p.role = 'teacher' and p.is_active = true and t.employment_status = 'active'
  ) then raise exception 'Assigned teacher must be active and belong to the same school.'; end if;
  return new;
end; $$;
create trigger class_subject_assignment_school_guard before insert or update on public.class_subjects
